using System;
using System.Collections.Generic;
using Neemle.OrmLite;
namespace Probe;
[Table("probe_authors",Managed=false)]
[RelationLoadLimits(MaxDepth=8,MaxPaths=16,MaxNodes=32,MaxRoots=100,MaxRelatedRows=1000,MaxStatements=64)]
public class Author {
    public int Code { get; set; }
    public string? Name { get; set; }
    [RelationLoad(RelationLoadMode.Eager)] public List<Book> Books { get; set; } = new();
}
[Table("probe_books",Managed=false)]
public class Book { public int Code { get; set; } public int AuthorId { get; set; }
    public Author? Author { get; set; } }
